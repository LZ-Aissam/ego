# EGO — Livraison E « Défis » Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Défis de groupe limités dans le temps (XP, séances, volume, force) avec classement en direct, enjeu écrit et classement final figé.

**Architecture:** Modèle `Challenge` + pivot des participants ; `ChallengeService` (règles), `ChallengeBoard` (scores à la volée), `ChallengeClock` (démarrage/clôture et messages, appelé par le middleware des quêtes). UI dans l'onglet Crew « Défis », page `/crew/defis/{challenge}` et une ligne sur l'accueil.

**Tech Stack:** Laravel 12, Livewire 3, Alpine, Tailwind 4 (`x-sys.*`), Pest 4 sur MariaDB `ego_testing`.

**Spec:** `docs/superpowers/specs/2026-10-09-ego-defis-design.md`

## Global Constraints

- MariaDB partout ; fuseau `Europe/Paris` ; départ le lendemain 00:00 ; fin exclue.
- Types : `xp`, `sessions`, `volume`, `strength` (+ `lift` ∈ `bench`, `squat`, `deadlift`). Durées : `3d`, `1w`, `2w`, `1m` (mois calendaire, sans débordement).
- Statuts : `upcoming`, `active`, `finished`, `cancelled`. Au plus 3 défis `upcoming`/`active` par groupe.
- Pas d'XP ni de titre : enjeu facultatif (≤ 100 caractères, affiché échappé) + classement final figé.
- Messages exacts : « Ce groupe a déjà 3 défis en cours. », « Ce défi est terminé. », « Le défi a commencé : tu ne peux plus le quitter. », « DÉFI LANCÉ », « DÉFI TERMINÉ ».
- Valeurs : `1 240 XP`, `4 séances`, `8 450 kg`, `+6 %` (espace insécable U+202F pour les milliers).
- Non-membre du groupe → 404 ; action non autorisée → 403 ; textes en français, style Système, « TON EGO ».

## Review Focus

1. Défi créé à 23:59 : départ le lendemain 00:00, pas le surlendemain (test Task 1).
2. Défi d'un mois démarrant un 31 janvier : fin au 28 février 00:00, sans débordement en mars (test Task 1).
3. Participant sans aucune séance pendant le défi : classé avec 0 en bas (XP, séances, volume), pas exclu (test Task 3).
4. Un participant n'ouvre l'app que 3 jours après la fin : tous les participants reçoivent « DÉFI TERMINÉ » une seule fois (test Task 4).
5. Groupe supprimé : ses défis disparaissent (cascade) sans erreur dans l'onglet Défis ni sur l'accueil (test Task 5).

```php
<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Illuminate\Database\Eloquent\Relations\BelongsToMany;
use Illuminate\Support\Carbon;

class Challenge extends Model
{
    public const METRICS = ['xp' => 'XP', 'sessions' => 'Séances', 'volume' => 'Volume', 'strength' => 'Force'];

    public const DURATIONS = ['3d' => '3 jours', '1w' => '1 semaine', '2w' => '2 semaines', '1m' => '1 mois'];

    public const LIFTS = ['bench' => 'Développé couché', 'squat' => 'Squat', 'deadlift' => 'Soulevé de terre'];

    protected $fillable = ['crew_id', 'creator_id', 'metric', 'lift', 'duration', 'starts_at', 'ends_at', 'stake', 'status', 'final_standings', 'started_notified_at', 'finished_notified_at'];

    protected function casts(): array
    {
        return [
            'starts_at' => 'datetime', 'ends_at' => 'datetime', 'final_standings' => 'array',
            'started_notified_at' => 'datetime', 'finished_notified_at' => 'datetime',
        ];
    }

    /** Départ le lendemain à 00:00, fin exclue. @return array{0: Carbon, 1: Carbon} */
    public static function window(string $duration, Carbon $now): array
    {
        $start = $now->copy()->addDay()->startOfDay();
        $end = match ($duration) {
            '3d' => $start->copy()->addDays(3),
            '1w' => $start->copy()->addWeek(),
            '2w' => $start->copy()->addWeeks(2),
            '1m' => $start->copy()->addMonthNoOverflow(),
        };

        return [$start, $end];
    }

    public function title(): string
    {
        return match ($this->metric) {
            'xp' => 'Défi XP',
            'sessions' => 'Défi séances',
            'volume' => 'Défi volume',
            'strength' => 'Défi force · '.(self::LIFTS[$this->lift] ?? ''),
        };
    }

    public function isOpen(): bool
    {
        return in_array($this->status, ['upcoming', 'active'], true);
    }

    public function crew(): BelongsTo
    {
        return $this->belongsTo(Crew::class);
    }

    public function creator(): BelongsTo
    {
        return $this->belongsTo(User::class, 'creator_id');
    }

    public function participants(): BelongsToMany
    {
        return $this->belongsToMany(User::class)->withPivot('joined_at');
    }

    public function hasParticipant(User $user): bool
    {
        return $this->participants()->whereKey($user->id)->exists();
    }
}
```

```php
<?php

namespace App\Crew;

use App\Models\Challenge;
use App\Models\Crew;
use App\Models\User;
use Illuminate\Support\Facades\DB;
use Illuminate\Validation\Rule;
use Illuminate\Validation\ValidationException;

final class ChallengeService
{
    public const MAX_OPEN = 3;

    public function create(User $user, Crew $crew, string $metric, ?string $lift, string $duration, ?string $stake): Challenge
    {
        abort_unless($crew->hasMember($user), 404);
        $stake = $stake === null ? null : (trim($stake) === '' ? null : trim($stake));

        validator(compact('metric', 'lift', 'duration', 'stake'), [
            'metric' => ['required', Rule::in(array_keys(Challenge::METRICS))],
            'lift' => [Rule::requiredIf($metric === 'strength'), 'nullable', Rule::in(array_keys(Challenge::LIFTS))],
            'duration' => ['required', Rule::in(array_keys(Challenge::DURATIONS))],
            'stake' => ['nullable', 'string', 'max:100'],
        ], [
            'metric.*' => 'Choisis un type de défi.',
            'lift.*' => 'Choisis le mouvement du défi de force.',
            'duration.*' => 'Choisis une durée.',
            'stake.max' => "L'enjeu doit faire au plus 100 caractères.",
        ])->validate();

        if ($crew->challenges()->whereIn('status', ['upcoming', 'active'])->count() >= self::MAX_OPEN) {
            throw ValidationException::withMessages(['crew' => 'Ce groupe a déjà 3 défis en cours.']);
        }

        [$start, $end] = Challenge::window($duration, now());

        return DB::transaction(function () use ($user, $crew, $metric, $lift, $duration, $stake, $start, $end) {
            $challenge = $crew->challenges()->create([
                'creator_id' => $user->id, 'metric' => $metric, 'lift' => $metric === 'strength' ? $lift : null,
                'duration' => $duration, 'starts_at' => $start, 'ends_at' => $end, 'stake' => $stake, 'status' => 'upcoming',
            ]);
            $challenge->participants()->attach($user->id, ['joined_at' => now()]);

            return $challenge;
        });
    }

    public function join(User $user, Challenge $challenge): void
    {
        abort_unless($challenge->crew->hasMember($user), 404);
        if (! $challenge->isOpen()) {
            throw ValidationException::withMessages(['challenge' => 'Ce défi est terminé.']);
        }
        if (! $challenge->hasParticipant($user)) {
            $challenge->participants()->attach($user->id, ['joined_at' => now()]);
        }
    }

    public function leave(User $user, Challenge $challenge): void
    {
        abort_unless($challenge->crew->hasMember($user), 404);
        abort_if($challenge->creator_id === $user->id, 403);
        if ($challenge->status !== 'upcoming') {
            throw ValidationException::withMessages(['challenge' => 'Le défi a commencé : tu ne peux plus le quitter.']);
        }
        $challenge->participants()->detach($user->id);
    }

    public function cancel(User $user, Challenge $challenge): void
    {
        abort_unless($challenge->creator_id === $user->id, 403);
        if ($challenge->status !== 'upcoming') {
            throw ValidationException::withMessages(['challenge' => 'Le défi a commencé : il ne peut plus être annulé.']);
        }
        $challenge->update(['status' => 'cancelled']);
    }

    /** Départ du groupe : retrait des défis du groupe qui ne sont pas terminés. */
    public function leaveCrew(User $user, Crew $crew): void
    {
        $crew->challenges()->whereIn('status', ['upcoming', 'active'])->get()
            ->each(fn (Challenge $c) => $c->participants()->detach($user->id));
    }
}
```

```php
<?php

namespace App\Crew;

use App\Models\Challenge;
use App\Models\Exercise;
use App\Models\User;
use App\Models\Workout;
use App\Models\WorkoutSet;
use App\Models\XpEvent;
use App\Services\RecordCalculator;
use Illuminate\Support\Carbon;
use Illuminate\Support\Collection;

final class ChallengeBoard
{
    public function __construct(private RecordCalculator $calc) {}

    /** Classement figé pour un défi terminé, sinon calculé en direct. */
    public function standings(Challenge $challenge): Collection
    {
        return $challenge->status === 'finished' && $challenge->final_standings !== null
            ? collect($challenge->final_standings)
            : $this->compute($challenge);
    }

    /** @return Collection<int, array{user_id: int, first_name: string, value: int|float, rank: int}> */
    public function compute(Challenge $challenge): Collection
    {
        $users = $challenge->participants()->get();
        $from = $challenge->starts_at;
        $to = $challenge->ends_at;

        $values = match ($challenge->metric) {
            'xp' => $this->xp($users, $from, $to),
            'sessions' => $this->workouts($users, $from, $to)->map->count(),
            'volume' => $this->workouts($users, $from, $to)->map(fn (Collection $ws) => round((float) $ws->sum(fn (Workout $w) => $w->volume()), 1)),
            'strength' => $this->strength($users, $challenge->lift, $from),
        };

        if ($challenge->metric !== 'strength') {
            $zero = $challenge->metric === 'volume' ? 0.0 : 0;
            $values = $users->mapWithKeys(fn (User $u) => [$u->id => $values[$u->id] ?? $zero]);
        }

        $rows = $users->filter(fn (User $u) => $values->has($u->id))
            ->sort(fn (User $a, User $b) => [$values[$b->id], Leaderboard::sortKey($a), $a->id] <=> [$values[$a->id], Leaderboard::sortKey($b), $b->id])
            ->values();

        $previous = null;
        $rank = 0;

        return $rows->map(function (User $u, int $i) use ($values, &$previous, &$rank) {
            if ($previous === null || $values[$u->id] != $previous) {
                $rank = $i + 1;
            }
            $previous = $values[$u->id];

            return ['user_id' => $u->id, 'first_name' => $u->first_name, 'value' => $values[$u->id], 'rank' => $rank];
        });
    }

    public static function format(string $metric, int|float $value): string
    {
        return match ($metric) {
            'xp' => number_format($value, 0, ',', "\u{202F}").' XP',
            'sessions' => $value.($value > 1 ? ' séances' : ' séance'),
            'volume' => number_format($value, 0, ',', "\u{202F}").' kg',
            'strength' => ($value < 0 ? '−' : '+').abs($value).' %',
        };
    }

    public static function remaining(Carbon $end): string
    {
        $minutes = max(0, (int) floor(now()->diffInMinutes($end, false)));
        $days = intdiv($minutes, 1440);
        $hours = intdiv($minutes % 1440, 60);

        return match (true) {
            $days > 0 => "encore {$days} j {$hours} h",
            $hours > 0 => "encore {$hours} h",
            default => 'encore '.$minutes.' min',
        };
    }

    private function xp(Collection $users, Carbon $from, Carbon $to): Collection
    {
        return XpEvent::whereIn('user_id', $users->modelKeys())->where('created_at', '>=', $from)->where('created_at', '<', $to)
            ->groupBy('user_id')->selectRaw('user_id, SUM(amount) AS total')->pluck('total', 'user_id')->map(fn ($v) => (int) $v);
    }

    private function workouts(Collection $users, Carbon $from, Carbon $to): Collection
    {
        return Workout::with('sets.exercise')->whereIn('user_id', $users->modelKeys())
            ->where('finished_at', '>=', $from)->where('finished_at', '<', $to)->get()->groupBy('user_id');
    }

    /** Gain en % du meilleur 1RM estimé pendant le défi sur le meilleur d'avant ; non classé sans l'un des deux. */
    private function strength(Collection $users, ?string $lift, Carbon $from): Collection
    {
        $exerciseId = Exercise::where('slug', Leaderboard::LIFTS[$lift] ?? 'bench-press')->value('id');
        $sets = WorkoutSet::with('workout', 'exercise')->where('exercise_id', $exerciseId)
            ->whereHas('workout', fn ($q) => $q->whereIn('user_id', $users->modelKeys()))->get()
            ->groupBy(fn (WorkoutSet $s) => $s->workout->user_id);

        return $users->mapWithKeys(function (User $u) use ($sets, $from) {
            $mine = $sets[$u->id] ?? collect();
            $best = fn (Collection $list) => $list->max(fn (WorkoutSet $s) => $this->calc->estimatedOneRepMax(
                $this->calc->effectiveLoad($s->weight_kg, $s->exercise->is_bodyweight, $s->workout->bodyweight_kg), $s->reps,
            ));
            $before = $best($mine->filter(fn (WorkoutSet $s) => $s->workout->finished_at->lt($from)));
            $during = $best($mine->filter(fn (WorkoutSet $s) => $s->workout->finished_at->gte($from)));

            return $before && $during ? [$u->id => (int) round(($during - $before) / $before * 100)] : [];
        });
    }
}
```

```php
<?php

namespace App\Crew;

use App\Models\Challenge;
use App\Models\User;
use App\Support\SystemMessage;
use App\Support\SystemNotifier;
use Illuminate\Support\Facades\DB;

/** Fait avancer les défis du joueur (début, clôture) et prévient chaque participant une seule fois. */
final class ChallengeClock
{
    public function __construct(private ChallengeBoard $board, private SystemNotifier $notifier) {}

    public function tick(User $user): void
    {
        $user->challenges()->whereIn('status', ['upcoming', 'active'])->where('starts_at', '<=', now())->pluck('challenges.id')
            ->each(fn (int $id) => $this->advance($id));
    }

    private function advance(int $id): void
    {
        DB::transaction(function () use ($id) {
            $challenge = Challenge::with('crew', 'participants')->lockForUpdate()->find($id);
            if (! $challenge || ! $challenge->isOpen()) {
                return;
            }

            if ($challenge->status === 'upcoming' && $challenge->starts_at->lte(now())) {
                $challenge->status = 'active';
                if (! $challenge->started_notified_at) {
                    $challenge->participants->each(fn (User $u) => $this->notifier->push($u, SystemMessage::challengeStarted($challenge)));
                    $challenge->started_notified_at = now();
                }
            }

            if ($challenge->status === 'active' && $challenge->ends_at->lte(now())) {
                $standings = $this->board->compute($challenge)->values();
                $challenge->final_standings = $standings->all();
                $challenge->status = 'finished';
                if (! $challenge->finished_notified_at) {
                    $winners = $standings->where('rank', 1)->pluck('first_name')->join(', ', ' et ') ?: 'personne';
                    foreach ($challenge->participants as $u) {
                        $rank = $standings->firstWhere('user_id', $u->id)['rank'] ?? null;
                        $this->notifier->push($u, SystemMessage::challengeFinished($challenge, $rank, $challenge->participants->count(), $winners));
                    }
                    $challenge->finished_notified_at = now();
                }
            }

            $challenge->save();
        });
    }
}
```

```php
<?php

namespace App\Livewire\Crew;

use App\Crew\ChallengeBoard;
use App\Crew\ChallengeService;
use App\Models\Challenge;
use Livewire\Attributes\Locked;
use Livewire\Attributes\Title;
use Livewire\Component;

#[Title('Défi')]
class ChallengePage extends Component
{
    #[Locked]
    public int $challengeId;

    public function mount(Challenge $challenge): void
    {
        abort_unless($challenge->crew->hasMember(auth()->user()), 404);
        $this->challengeId = $challenge->id;
    }

    private function challenge(): Challenge
    {
        $challenge = Challenge::with('crew')->find($this->challengeId);
        abort_unless($challenge && $challenge->crew->hasMember(auth()->user()), 404);

        return $challenge;
    }

    public function join(ChallengeService $service): void
    {
        $service->join(auth()->user(), $this->challenge());
    }

    public function leave(ChallengeService $service): void
    {
        $service->leave(auth()->user(), $this->challenge());
    }

    public function cancel(ChallengeService $service)
    {
        $service->cancel(auth()->user(), $this->challenge());

        return $this->redirectRoute('community', ['tab' => 'defis']);
    }

    public function render(ChallengeBoard $board)
    {
        $challenge = $this->challenge();

        return view('livewire.crew.challenge-page', [
            'challenge' => $challenge,
            'standings' => $board->standings($challenge),
            'participating' => $challenge->hasParticipant(auth()->user()),
            'isCreator' => $challenge->creator_id === auth()->id(),
        ]);
    }
}
```
